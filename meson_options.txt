       description: 'Build with OpenGL support')

       description: 'Build with EGL support (requires GL)')

       description: 'Build with X11 support (requires GL)')

option('wayland', type: 'feature', value: 'auto',
       description: 'Build with Wayland support (requires EGL)')

option('vulkan', type: 'feature', value: 'auto',
       description: 'Build with Vulkan support')

       description: 'Build with Metal support (macOS/iOS only)')

       description: 'Enable SVG rendering support')

       description: 'Enable PDF rendering support')

       description: 'Enable WebP encoding support')

       description: 'Enable WebP decoding support')
